cmake_minimum_required(VERSION 3.16)
project(albert_recovered_constants LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(albert_recovered src/board_profile.c src/http_routes.c src/summary.c)
target_include_directories(albert_recovered PUBLIC include)
target_compile_options(albert_recovered PRIVATE -Wall -Wextra -Werror)
add_executable(albert_inspect src/main.c)
target_link_libraries(albert_inspect PRIVATE albert_recovered)
